import importlib.util
from pathlib import Path

dataset_path = Path('..') / 'Lab1' / 'data' / 'Datos Lab 1.csv'
print('Dataset:', dataset_path.resolve())

module_path = Path('implementacion_lab2.py').resolve()
spec = importlib.util.spec_from_file_location('implementacion_lab2', module_path)
impl = importlib.util.module_from_spec(spec)
spec.loader.exec_module(impl)
print('Implementación cargada:', module_path.name)
